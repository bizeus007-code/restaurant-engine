### 1. ADIM: QRALL LİNKİNDEN CANLI MENÜ VERİLERİNİ ÇEK VE AYIKLA (BASH / NODE SCRIPT)
Terminalde aşağıdaki Node.js betiğini çalıştırarak Beroş Restaurant'ın QRall menüsündeki tüm gerçek kategorileri, ürün adlarını, açıklamalarını, fiyatlarını ve görsellerini çekip `scratch/qrall-menu.json` dosyasına kaydet:

```bash
mkdir -p scratch

node -e '
const https = require("https");
const fs = require("fs");

const targetUrl = "[https://view.qrall.co/tr/categories?tenantId=3a17ab56-5e5c-eb02-fbd8-5d5462e53f13&menu=3a17ab56-71f1-d4f3-6f76-1fd1d8aa9402&channelId=3a17ab56-616b-5213-d3df-231483a5fe8d](https://view.qrall.co/tr/categories?tenantId=3a17ab56-5e5c-eb02-fbd8-5d5462e53f13&menu=3a17ab56-71f1-d4f3-6f76-1fd1d8aa9402&channelId=3a17ab56-616b-5213-d3df-231483a5fe8d)";

https.get(targetUrl, { headers: { "User-Agent": "Mozilla/5.0 (Macintosh; Intel Mac OS X 10_15_7)" } }, (res) => {
  let html = "";
  res.on("data", chunk => html += chunk);
  res.on("end", async () => {
    // 1. Next.js __NEXT_DATA__ JSON kontrolü
    const nextDataMatch = html.match(/<script id="__NEXT_DATA__" type="application\/json">([^<]+)<\/script>/);
    if (nextDataMatch) {
      try {
        const nextData = JSON.parse(nextDataMatch[1]);
        const pageProps = nextData.props?.pageProps || {};
        fs.writeFileSync("scratch/qrall-menu.json", JSON.stringify(pageProps, null, 2));
        console.log("QRall Menü Verisi __NEXT_DATA__ üzerinden başarıyla çıkarıldı.");
        return;
      } catch (e) {
        console.log("JSON parse hatası:", e.message);
      }
    }

    // 2. Yedek API veya HTML Ayrıştırma
    console.log("HTML uzunluğu:", html.length);
    fs.writeFileSync("scratch/qrall-raw.html", html);
    console.log("Ham HTML scratch/qrall-raw.html dosyasına yazıldı.");
  });
}).on("error", err => {
  console.error("QRall veri çekme hatası:", err.message);
});
'
```
